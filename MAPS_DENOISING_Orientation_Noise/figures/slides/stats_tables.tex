\begin{table}[h]
\centering
\caption{Statistical test: our network against the best MTEX filter}
\begin{tabular}{>{\bfseries}l p{16cm}}
\toprule
Compared & Our network vs.\ the MTEX filter with the lowest error on that measure and test set \\
Per map  & $d_m = e_m^{\text{ours}} - e_m^{\text{MTEX}}$, \quad $m = 1, \dots, 10$ \quad (same maps, same noise) \\
$\boldsymbol{H_0}$ & $\mu_d = 0$: our network and the best MTEX filter have the same mean error \\
$\boldsymbol{H_1}$ & $\mu_d \neq 0$: the mean errors differ (two-sided) \\
Test     & Paired $t$-test, $n = 10$ maps, 9 degrees of freedom, Holm correction over the test sets \\
Decision & Reject $H_0$ if $p < 0.05$; ours is better if $\bar{d} < 0$, worse if $\bar{d} > 0$ \\
\bottomrule
\end{tabular}
\end{table}
